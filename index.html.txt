    <!-- JavaScript Ana Kod Yapısı -->
    <script>
        // Otopark Mock Verileri
        const parkingData = [
            { id: 1, name: "Kadıköy İSPARK Katlı", type: "ispark", lat: 40.9901, lng: 29.0291, capacity: 250, empty: 42, price: 50, ev: true },
            { id: 2, name: "Beşiktaş Meydan Otoparkı", type: "ispark", lat: 41.0422, lng: 29.0067, capacity: 180, empty: 8, price: 60, ev: false },
            { id: 3, name: "Moda Caddesi Yol Kenarı", type: "street", lat: 40.9845, lng: 29.0255, capacity: 30, empty: 5, price: 0, ev: false },
            { id: 4, name: "Nişantaşı City's Otoparkı", type: "private", lat: 41.0485, lng: 28.9942, capacity: 400, empty: 120, price: 100, ev: true }
        ];

        let map, markers = [];

        // Haritayı Başlat
        function initMap() {
            map = L.map('map').setView([41.0082, 28.9784], 12);
            L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
                attribution: '&copy; OpenStreetMap'
            }).addTo(map);

            renderMarkers(parkingData);
            renderStreetPolylines();
            renderList(parkingData);
        }

        // Markerları Haritaya Ekle
        function renderMarkers(data) {
            markers.forEach(m => map.removeLayer(m));
            markers = [];

            data.forEach(item => {
                const color = item.empty > 20 ? '#10b981' : (item.empty > 0 ? '#f59e0b' : '#ef4444');
                const marker = L.circleMarker([item.lat, item.lng], {
                    radius: 10,
                    fillColor: color,
                    color: '#fff',
                    weight: 2,
                    opacity: 1,
                    fillOpacity: 0.8
                }).addTo(map);

                marker.bindPopup(`
                    <div style="color: #000; font-family: sans-serif;">
                        <h4 style="font-weight: bold; margin: 0 0 4px 0;">${item.name}</h4>
                        <p style="margin: 0; font-size: 12px;">Boş Yer: <b>${item.empty}</b> / ${item.capacity}</p>
                        <p style="margin: 0; font-size: 12px;">Saatlik: ₺${item.price}</p>
                        <button onclick="setMyCar('${item.name}')" style="margin-top: 6px; background: #0284c7; color: #fff; border: none; padding: 4px 8px; border-radius: 4px; font-size: 11px; cursor: pointer;">Buraya Park Et</button>
                    </div>
                `);

                markers.push(marker);
            });
        }

        // Yol Kenarı Polylines Çizimi
        function renderStreetPolylines() {
            const streetCoords = [
                [[40.9855, 29.0265], [40.9835, 29.0245]],
                [[41.0432, 29.0077], [41.0412, 29.0057]]
            ];

            streetCoords.forEach(coords => {
                L.polyline(coords, { color: '#10b981', weight: 5, opacity: 0.7 }).addTo(map);
            });
        }

        // Liste Renderleme
        function renderList(data) {
            const container = document.getElementById('parkingList');
            container.innerHTML = '';

            data.forEach(item => {
                const badgeColor = item.empty > 20 ? 'text-emerald-400 bg-emerald-500/10' : 'text-amber-400 bg-amber-500/10';
                container.innerHTML += `
                    <div class="p-3 bg-slate-800/50 border border-slate-700/60 rounded-xl hover:border-sky-500/50 transition cursor-pointer" onclick="map.setView([${item.lat}, ${item.lng}], 15)">
                        <div class="flex justify-between items-start">
                            <h3 class="font-bold text-sm text-slate-200">${item.name}</h3>
                            <span class="text-xs px-2 py-0.5 rounded-full font-semibold ${badgeColor}">${item.empty} Boş</span>
                        </div>
                        <div class="flex items-center gap-3 text-xs text-slate-400 mt-2">
                            <span><i class="fa-solid fa-turkish-lira-sign"></i> ${item.price}/saat</span>
                            ${item.ev ? '<span class="text-sky-400"><i class="fa-solid fa-bolt"></i> EV Var</span>' : ''}
                        </div>
                    </div>
                `;
            });
        }

        // Arabam Nerede Takibi
        function setMyCar(name) {
            document.getElementById('carLocationName').innerText = name;
            document.getElementById('myCarWidget').classList.remove('hidden');
            alert(`${name} aracınızın park konumu olarak kaydedildi!`);
        }

        function clearMyCar() {
            document.getElementById('myCarWidget').classList.add('hidden');
        }

        // Modal Yönetimi
        function openModal(id) { document.getElementById(id).classList.remove('hidden'); }
        function closeModal(id) { document.getElementById(id).classList.add('hidden'); }
        function toggleCarPlay() { document.getElementById('carPlayOverlay').classList.toggle('hidden'); }

        window.onload = initMap;
    </script>
